  <script>
    const DB_NAME = 'FitBotDB';
    const DB_VERSION = 1;
    let db;

    function initDB() {
      return new Promise((resolve, reject) => {
        const request = indexedDB.open(DB_NAME, DB_VERSION);
        request.onupgradeneeded = (e) => {
          const db = e.target.result;
          if (!db.objectStoreNames.contains('store')) {
            db.createObjectStore('store');
          }
        };
        request.onsuccess = (e) => {
          db = e.target.result;
          resolve(db);
        };
        request.onerror = (e) => reject(e);
      });
    }

    async function saveData(key, val) {
      if (!db) await initDB();
      return new Promise((resolve) => {
        const tx = db.transaction('store', 'readwrite');
        tx.objectStore('store').put(val, key);
        tx.oncomplete = () => resolve();
      });
    }

    async function getData(key, defaultVal) {
      if (!db) await initDB();
      return new Promise((resolve) => {
        const tx = db.transaction('store', 'readonly');
        const req = tx.objectStore('store').get(key);
        req.onsuccess = () => resolve(req.result !== undefined ? req.result : defaultVal);
        req.onerror = () => resolve(defaultVal);
      });
    }

      workout: 'Giorno A (Petto/Tricipiti):\n- Panca Piana 4x8\n- Spinte Manubri Inclinata 3x10\n- Dip Parallele 3xMax\n- French Press 3x12\n\nGiorno B (Dorso/Bicipiti):\n- Trazioni Sbarra 4x6\n- Rematore Bilanciere 4x8\n- Pulley Basso 3x10\n- Curl Bilanciere 3x10\n\nGiorno C (Gambe/Spalle):\n- Squat 4x8\n- Pressa 3x10\n- Military Press 4x8\n- Alzate Laterali 4x12',

    function bindEvents() {
      document.querySelectorAll('.nav-btn').forEach(btn => {
        btn.addEventListener('click', () => {
          const tab = btn.dataset.tab;
          document.querySelectorAll('.nav-btn').forEach(b => {
            b.classList.remove('active', 'text-blue-400');
            b.classList.add('text-slate-400');
          });
          btn.classList.add('active', 'text-blue-400');
          btn.classList.remove('text-slate-400');

          document.querySelectorAll('.tab-content').forEach(c => c.classList.add('hidden'));
          document.getElementById(`tab-${tab}`).classList.remove('hidden');
        });
      });

      const saveInput = (id, key) => {
        document.getElementById(id).addEventListener('input', (e) => {
          appState[key] = e.target.value;
          saveData(key, e.target.value);
        });
      };

      saveInput('profWeight', 'weight');
      saveInput('profHeight', 'height');
      saveInput('profGoal', 'goal');
      saveInput('profNotes', 'notes');
      saveInput('dietData', 'diet');
      saveInput('workoutData', 'workout');

      chatForm.addEventListener('submit', (e) => {
        e.preventDefault();
        const text = chatInput.value.trim();
        if (text) {
          handleUserQuery(text);
          chatInput.value = '';
        }
      });

      if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {
        const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
        const recognition = new SpeechRecognition();
        recognition.lang = 'it-IT';

        voiceBtn.addEventListener('click', () => {
          recognition.start();
          voiceBtn.classList.add('text-red-500', 'animate-pulse');
        });

        recognition.onresult = (e) => {
          const transcript = e.results[0][0].transcript;
          handleUserQuery(transcript);
        };

        recognition.onend = () => {
          voiceBtn.classList.remove('text-red-500', 'animate-pulse');
        };
      }

      document.getElementById('openSettings').addEventListener('click', () => {
        document.getElementById('settingsModal').classList.remove('hidden');
      });

      document.getElementById('closeSettings').addEventListener('click', () => {
        document.getElementById('settingsModal').classList.add('hidden');
      });

      document.getElementById('ttsToggle').addEventListener('change', (e) => {
        appState.ttsEnabled = e.target.checked;
        saveData('ttsEnabled', e.target.checked);
      });
    }

    async function handleUserQuery(text) {
      addMessage('user', text);

      let reply = '';
      const lower = text.toLowerCase();

      if (lower.includes('sostituisci') || lower.includes('alternativa') || lower.includes('cambiare')) {
        for (let food in FOOD_SUBSTITUTES) {
          if (lower.includes(food)) {
            reply = `Sostituzioni per **${food}** basate sul tuo piano alimentare:\n`;
            FOOD_SUBSTITUTES[food].forEach(sub => {
              reply += `• **${sub.name}**: moltiplica il peso x${sub.ratio} per pari apporto.\n`;
            });
            break;
          }
        }

        if (!reply) {
          for (let ex in EXERCISE_SUBSTITUTES) {
            if (lower.includes(ex)) {
              reply = `Esercizi alternativi per **${ex}**:\n`;
              EXERCISE_SUBSTITUTES[ex].forEach(s => {
                reply += `• ${s}\n`;
              });
              break;
            }
          }
        }
      }

      if (!reply) {
        reply = `Ho registrato: "${text}". Rispetto ai tuoi dati (${appState.weight}kg, obiettivo ${appState.goal}), mantieni costanti i macronutrienti nei pasti principali.`;
      }

      addMessage('bot', reply);

      if (appState.ttsEnabled && 'speechSynthesis' in window) {
        const utterance = new SpeechSynthesisUtterance(reply.replace(/\*/g, ''));
        utterance.lang = 'it-IT';
        window.speechSynthesis.speak(utterance);
      }
    }

    function addMessage(role, text) {
      appState.messages.push({ role, text });
      saveData('messages', appState.messages);
      renderMessages();
    }

    function renderMessages() {
      chatMessages.innerHTML = '';
      appState.messages.forEach(m => {
        const isUser = m.role === 'user';
        const div = document.createElement('div');
        div.className = `flex gap-3 max-w-[85%] ${isUser ? 'ml-auto flex-row-reverse' : ''}`;
        
        div.innerHTML = `
          <div class="w-8 h-8 rounded-lg ${isUser ? 'bg-blue-600' : 'bg-blue-600/20 text-blue-400 border border-blue-500/30'} flex items-center justify-center shrink-0">
            <i data-lucide="${isUser ? 'user' : 'bot'}" class="w-4 h-4 text-white"></i>
          </div>
          <div class="glass-card p-3.5 rounded-2xl ${isUser ? 'rounded-tr-none bg-blue-600/20 border-blue-500/30' : 'rounded-tl-none'} text-sm leading-relaxed text-slate-200 whitespace-pre-line">
            ${m.text}
          </div>
        `;
        chatMessages.appendChild(div);
      });
      lucide.createIcons();
      chatMessages.scrollTop = chatMessages.scrollHeight;
    }
